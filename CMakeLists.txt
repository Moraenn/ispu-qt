cmake_minimum_required(VERSION 3.16)
project(ISPUQtLabs LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
option(BUILD_GUI "Build Qt applications" ON)
include(CTest)
add_library(studyplan_core StudyPlanner/core/studyplan.cpp)
target_include_directories(studyplan_core PUBLIC StudyPlanner/core)
if(MSVC)
    target_compile_options(studyplan_core PUBLIC /utf-8)
else()
    target_compile_options(studyplan_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING)
    add_executable(domain_tests tests/domain_tests.cpp)
    target_link_libraries(domain_tests PRIVATE studyplan_core)
    add_test(NAME domain_tests COMMAND domain_tests)
endif()
if(BUILD_GUI)
    find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Widgets)
    find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Widgets)
    set(CMAKE_AUTOMOC ON)
    if(MSVC)
        add_compile_options(/utf-8)
    endif()
    add_subdirectory(Calculator)
    add_subdirectory(Alarm)
    add_subdirectory(handbook_lab3)
    add_subdirectory(handbook)
    add_subdirectory(StudyPlanner)
    if(BUILD_TESTING)
        find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Test)
        add_subdirectory(tests/qt)
    endif()
endif()
